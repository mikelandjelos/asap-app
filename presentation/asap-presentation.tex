\begin{frame}{Problem i ideja}
  \begin{itemize}
    \item Podaci o proizvodima su rasuti, a korisniku su potrebni odmah, u trenutku kupovine.
    \item ASAP povezuje skenirani barkod sa metapodacima proizvoda.
    \item Semantičke reprezentacije omogućavaju pretragu sličnih proizvoda i personalizovane preporuke.
  \end{itemize}
\end{frame}

\begin{frame}{Planirane AI funkcionalnosti}
  \begin{enumerate}
    \item lokalno prepoznavanje i dekodiranje barkoda;
    \item embedding reprezentacije proizvoda;
    \item top-N semantička pretraga kosinusnom sličnošću;
    \item opcioni MMR za raznovrsnije rezultate;
    \item preporuka iz ograničene istorije; tačan model profila bira se naknadno;
    \item opciona klasterizacija i PCA vizuelna analitika.
  \end{enumerate}
\end{frame}

\begin{frame}{Granica operativnog MVP-a}
  \begin{itemize}
    \item P0 jezgro (80): skeniranje, metapodaci sa rezervnim izvorom, detalji proizvoda, top-N semantička sličnost i ponovljiva demonstracija;
    \item svih 80 P0 poena čine jedan obavezni tok --- nisu proizvoljno zamenljivi;
    \item prošireni MVP, P1 (15): obavezna preporuka iz ograničene nedavne istorije, uz eksplicitno cold-start stanje;
    \item tačan broj interakcija i način ponderisanja/agregacije biraju se naknadno;
    \item P2 (5): šira evaluacija i povratne informacije;
    \item van MVP-a: nalozi, objavljivanje, prilagođena kamera, MMR, klasterizacija, PCA i produkciono skaliranje.
  \end{itemize}
\end{frame}

\begin{frame}{Plan tankih iteracija}
  \begin{enumerate}
    \item \textbf{I0:} skener --- implementiran i fizički potvrđen;
    \item \textbf{I1:} deterministički vertikalni tok sa kontrolisanim podacima;
    \item \textbf{I2:} stvarni metapodaci proizvoda i rezervni izvor;
    \item \textbf{I3:} embedding reprezentacije i top-N semantička sličnost;
    \item \textbf{I4:} prošireni MVP --- preporuka iz istorije i cold-start stanje;
    \item \textbf{I5:} stabilizacija, evaluacija i demonstracija.
  \end{enumerate}
\end{frame}

\section{Tehnologije i podaci}

\begin{frame}{Izabrane tehnologije}
  \begin{itemize}
    \item Android klijent: Java/XML modul \texttt{android/app}; build, sedam testova i lint prolaze;
    \item skeniranje: Google Code Scanner 16.1.0, EAN/UPC, bez dozvole aplikacije za kameru;
    \item build: AGP 9.3.2, Gradle 9.5.0, Java 17, API 36 i \texttt{minSdk 23};
    \item UI: AppCompat 1.8.0 i ConstraintLayout 2.2.2;
    \item test uređaj: Samsung SM-A566B, Android 16/API 36, Google Play servisi i potrebne kamere verifikovani preko ADB-a;
    \item prilagođeni skener sa CameraX i direktnim ML Kit API-jem samo ako se pokaže potrebnim;

  \end{itemize}
